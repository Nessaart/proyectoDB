CREATE DATABASE IF NOT EXISTS ticket_platform_db;
USE ticket_platform_db;

-- 1. Tabla de Usuarios
CREATE TABLE users (
  id INT AUTO_INCREMENT PRIMARY KEY,
  email VARCHAR(255) NOT NULL UNIQUE,
  password_hash VARCHAR(255) NOT NULL, -- Nota: En este corte se almacena en texto plano según la instrucción
  full_name VARCHAR(150) NOT NULL,
  role ENUM('Client', 'Admin') NOT NULL DEFAULT 'Client',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 2. Tabla de Eventos
CREATE TABLE events (
  id INT AUTO_INCREMENT PRIMARY KEY,
  name VARCHAR(150) NOT NULL,
  location VARCHAR(150) NOT NULL,
  date_time DATETIME NOT NULL,
  category ENUM('Concert', 'Sport', 'Theater') NOT NULL,
  min_price DECIMAL(10, 2) NOT NULL,
  max_price DECIMAL(10, 2) NOT NULL,
  description TEXT NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 3. Tabla de Entradas / Transacciones
CREATE TABLE tickets (
  id INT AUTO_INCREMENT PRIMARY KEY,
  user_id INT NOT NULL,
  event_id INT NOT NULL,
  purchase_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  paid_amount DECIMAL(10, 2) NOT NULL,
  status ENUM('Valid', 'Used', 'Cancelled') NOT NULL DEFAULT 'Valid',
  FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
  FOREIGN KEY (event_id) REFERENCES events(id) ON DELETE CASCADE
);

-- 4. Detalles Especializados (Conciertos)
CREATE TABLE event_concert_details (
  id INT AUTO_INCREMENT PRIMARY KEY,
  event_id INT NOT NULL UNIQUE,
  artist_lineup TEXT NOT NULL,
  FOREIGN KEY (event_id) REFERENCES events(id) ON DELETE CASCADE
);

-- 5. Detalles Especializados (Deportes)
CREATE TABLE event_sport_details (
  id INT AUTO_INCREMENT PRIMARY KEY,
  event_id INT NOT NULL UNIQUE,
  home_team VARCHAR(100) NOT NULL,
  away_team VARCHAR(100) NOT NULL,
  live_score VARCHAR(50) DEFAULT '0-0',
  FOREIGN KEY (event_id) REFERENCES events(id) ON DELETE CASCADE
);

-- 6. Detalles Especializados (Teatro)
CREATE TABLE event_theater_details (
  id INT AUTO_INCREMENT PRIMARY KEY,
  event_id INT NOT NULL UNIQUE,
  cast_members TEXT NOT NULL,
  intermissions_duration VARCHAR(50) NOT NULL,
  FOREIGN KEY (event_id) REFERENCES events(id) ON DELETE CASCADE
);

-- 7. Reseñas y Calificaciones
CREATE TABLE reviews (
  id INT AUTO_INCREMENT PRIMARY KEY,
  user_id INT NOT NULL,
  event_id INT NOT NULL,
  rating INT CHECK (rating BETWEEN 1 AND 5),
  comment TEXT,
  seat_valuation VARCHAR(50),
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
  FOREIGN KEY (event_id) REFERENCES events(id) ON DELETE CASCADE
);
